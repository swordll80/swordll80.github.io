SQL 数据库手册

18 条
关系型数据库 SQL 速查：查询、连接、聚合、增删改、索引与事务，语法以标准 SQL/MySQL 为主。
1 SELECT
SELECT id, name FROM users; SELECT * FROM users; SELECT DISTINCT city FROM users; -- 去重 SELECT name AS 姓名 FROM users; -- 别名
2 WHERE 条件
SELECT * FROM users WHERE age >= 18 AND city = 'BJ' AND name LIKE '张%' -- 张开头 AND age BETWEEN 20 AND 30 AND id IN (1,2,3) AND phone IS NOT NULL;
3 排序与分页
SELECT * FROM users ORDER BY age DESC, name ASC LIMIT 10 OFFSET 20; -- 跳过20取10 -- SQL Server: OFFSET 20 ROWS FETCH NEXT 10 ROWS ONLY
4 聚合函数
SELECT COUNT(*), SUM(salary), AVG(salary), MAX(age), MIN(age) FROM employees;
5 GROUP BY / HAVING
SELECT dept, COUNT(*) AS n, AVG(salary) AS avg_s FROM employees GROUP BY dept HAVING COUNT(*) > 5 -- 对分组结果过滤 ORDER BY n DESC;
6 执行顺序
-- 书写：SELECT ... FROM ... WHERE ... GROUP BY ... HAVING ... ORDER BY ... LIMIT -- 执行：FROM → WHERE → GROUP BY → HAVING → SELECT → ORDER BY → LIMIT
7 JOIN 类型
-- INNER JOIN：两表都匹配 SELECT u.name, o.amount FROM users u INNER JOIN orders o ON u.id = o.user_id; -- LEFT JOIN：保留左表全部，右表无则 NULL -- RIGHT JOIN：保留右表全部 -- FULL JOIN：两表全部（MySQL 不支持，用 UNION 模拟）
8 子查询
SELECT * FROM users WHERE id IN (SELECT user_id FROM orders WHERE amount>100); SELECT name, (SELECT COUNT(*) FROM orders o WHERE o.user_id=u.id) AS cnt FROM users u;
9 UNION
SELECT name FROM a UNION -- 去重合并 SELECT name FROM b; -- UNION ALL 不去重，更快
10 INSERT
INSERT INTO users (name, age) VALUES ('张三', 20); INSERT INTO users (name, age) VALUES ('李四', 22), ('王五', 25); -- 批量
11 UPDATE
UPDATE users SET age = age + 1 WHERE city = 'BJ'; -- 注意：不加 WHERE 会更新全表！
12 DELETE / TRUNCATE
DELETE FROM users WHERE id = 5; TRUNCATE TABLE logs; -- 清空全表，比 DELETE 快、不可回滚
13 建表 DDL
CREATE TABLE users ( id INT PRIMARY KEY AUTO_INCREMENT, name VARCHAR(50) NOT NULL, age INT DEFAULT 0, email VARCHAR(100) UNIQUE, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP );
14 修改表
ALTER TABLE users ADD COLUMN phone VARCHAR(20); ALTER TABLE users DROP COLUMN phone; ALTER TABLE users MODIFY name VARCHAR(80); DROP TABLE users;
15 索引
CREATE INDEX idx_name ON users(name); CREATE UNIQUE INDEX idx_email ON users(email); -- 索引加速查询，但拖慢写入、占空间 -- WHERE/JOIN/ORDER BY 的列适合建索引 EXPLAIN SELECT * FROM users WHERE name='张三'; -- 看执行计划
16 事务 ACID
BEGIN; -- 或 START TRANSACTION UPDATE acct SET bal=bal-100 WHERE id=1; UPDATE acct SET bal=bal+100 WHERE id=2; COMMIT; -- 提交；出错则 ROLLBACK -- ACID：原子性/一致性/隔离性/持久性
17 隔离级别
-- 由低到高： -- 读未提交 → 脏读 -- 读已提交 → 不可重复读 -- 可重复读 → 幻读（MySQL 默认） -- 串行化 → 最严格、最慢
18 常用约束
PRIMARY KEY 主键，唯一且非空 FOREIGN KEY 外键，引用其他表 UNIQUE 唯一 NOT NULL 非空 CHECK(age>0) 检查 DEFAULT 默认值
