常见数据结构、复杂度、经典算法思想速查，每种算法配 C/C++ 示例代码便于理解与背诵。
1 时间复杂度
O(1)常数 < O(log n)对数 < O(n)线性 < O(n log n) < O(n²) < O(2ⁿ) < O(n!)。 看数据翻倍时耗时如何增长;只保留最高阶、去掉系数。
2 空间复杂度
额外空间随规模的增长;递归要计入调用栈深度。 原地算法 O(1) 额外空间(如快排/堆排)。
3 快速排序 O(n log n)
// 分治+基准，原地，平均最快 void quick(int a[], int l, int r){ if(l>=r) return; int i=l, j=r, p=a[(l+r)/2]; while(i<=j){ while(a[i]<p) i++; while(a[j]>p) j--; if(i<=j) std::swap(a[i++], a[j--]); } quick(a,l,j); quick(a,i,r); }
4 归并排序 O(n log n) 稳定
void merge_sort(int a[], int l, int r, int t[]){ if(l>=r) return; int m=(l+r)/2; merge_sort(a,l,m,t); merge_sort(a,m+1,r,t); int i=l,j=m+1,k=l; while(i<=m&&j<=r) t[k++]= a[i]<=a[j]? a[i++]:a[j++]; while(i<=m) t[k++]=a[i++]; while(j<=r) t[k++]=a[j++]; for(int x=l;x<=r;x++) a[x]=t[x]; }
5 堆排序 O(n log n)
// 用大顶堆，原地 void heapify(int a[], int n, int i){ int m=i, l=2*i+1, r=2*i+2; if(l<n && a[l]>a[m]) m=l; if(r<n && a[r]>a[m]) m=r; if(m!=i){ std::swap(a[i],a[m]); heapify(a,n,m); } } void heap_sort(int a[], int n){ for(int i=n/2-1;i>=0;i--) heapify(a,n,i); for(int i=n-1;i>0;i--){ std::swap(a[0],a[i]); heapify(a,i,0); } }
6 冒泡/选择/插入 O(n²)
// 插入排序：近乎有序时快，稳定 void insertion(int a[], int n){ for(int i=1;i<n;i++){ int k=a[i], j=i-1; while(j>=0 && a[j]>k){ a[j+1]=a[j]; j--; } a[j+1]=k; } } // 稳定：归并/插入/冒泡；不稳定：快排/选择/堆
7 二分查找 O(log n)
// 前提：有序数组，返回下标或 -1 int bsearch(int a[], int n, int x){ int lo=0, hi=n-1; while(lo<=hi){ int mid=lo+(hi-lo)/2; // 防溢出 if(a[mid]==x) return mid; else if(a[mid]<x) lo=mid+1; else hi=mid-1; } return -1; }
8 二分求下界(lower_bound)
// 第一个 >= x 的位置 int lower(int a[], int n, int x){ int lo=0, hi=n; // [lo,hi) while(lo<hi){ int mid=(lo+hi)/2; if(a[mid]<x) lo=mid+1; else hi=mid; } return lo; } // 二分也可用于“答案单调”的最优化问题
9 双指针
// 有序数组两数之和 = target bool twoSum(int a[], int n, int t){ int i=0, j=n-1; while(i<j){ int s=a[i]+a[j]; if(s==t) return true; else if(s<t) i++; else j--; } return false; }
10 滑动窗口
// 无重复字符的最长子串 int longest(std::string s){ int last[128]; std::fill(last,last+128,-1); int best=0, start=0; for(int i=0;i<(int)s.size();i++){ if(last[s[i]]>=start) start=last[s[i]]+1; last[s[i]]=i; best=std::max(best, i-start+1); } return best; }
11 快慢指针
// 判断链表是否有环 bool hasCycle(Node* head){ Node *slow=head, *fast=head; while(fast && fast->next){ slow=slow->next; fast=fast->next->next; if(slow==fast) return true; } return false; }
12 递归三要素
// 终止条件 + 递推关系 + 返回值 long fact(int n){ if(n<=1) return 1; // 终止 return n * fact(n-1); // 递推 }

14 快速幂 O(log n)
long qpow(long a, long b, long mod){ long r=1; a%=mod; while(b){ if(b&1) r=r*a%mod; a=a*a%mod; b>>=1; } return r; }
15 DP 思路
适用:最优子结构 + 重叠子问题。 步骤:定义状态 → 转移方程 → 初始化 → 遍历顺序 → 答案。
16 斐波那契(滚动)
int fib(int n){ if(n<2) return n; int a=0, b=1; for(int i=2;i<=n;i++){ int c=a+b; a=b; b=c; } return b; }
17 0-1 背包
// n 件物品(重w[i],值v[i])，容量 W int knapsack(int n,int W,int w[],int v[]){ std::vector<int> dp(W+1,0); for(int i=0;i<n;i++) for(int j=W;j>=w[i];j--) // 逆序保证每件只选一次 dp[j]=std::max(dp[j], dp[j-w[i]]+v[i]); return dp[W]; }
18 最长递增子序列 LIS
// O(n log n) int lis(std::vector<int>& a){ std::vector<int> t; for(int x:a){ auto it=std::lower_bound(t.begin(),t.end(),x); if(it==t.end()) t.push_back(x); else *it=x; } return t.size(); }
19 最长公共子序列 LCS
int lcs(std::string a, std::string b){ int n=a.size(), m=b.size(); std::vector<std::vector<int>> dp(n+1, std::vector<int>(m+1,0)); for(int i=1;i<=n;i++) for(int j=1;j<=m;j++) dp[i][j]= a[i-1]==b[j-1]? dp[i-1][j-1]+1 : std::max(dp[i-1][j], dp[i][j-1]); return dp[n][m]; }
20 贪心思想
每步取当前局部最优，期望得到全局最优;需证明贪心选择性质。 例:找零、区间调度、霍夫曼编码、最小生成树。
21 区间调度(最多不重叠区间)
// 按结束时间排序，能选就选 int maxIntervals(std::vector<std::pair<int,int>>& v){ std::sort(v.begin(),v.end(),[](auto&a,auto&b){return a.second<b.second;}); int cnt=0, end=-1e9; for(auto& p:v) if(p.first>=end){ cnt++; end=p.second; } return cnt; }
22 回溯模板
// 选择 → 递归 → 撤销选择 // 全排列 void permute(std::vector<int>& a, int k, std::vector<std::vector<int>>& res){ if(k==(int)a.size()){ res.push_back(a); return; } for(int i=k;i<(int)a.size();i++){ std::swap(a[k],a[i]); // 选择 permute(a,k+1,res); // 递归 std::swap(a[k],a[i]); // 撤销 } }
23 子集
void subsets(std::vector<int>& a,int k,std::vector<int>& cur, std::vector<std::vector<int>>& res){ if(k==(int)a.size()){ res.push_back(cur); return; } subsets(a,k+1,cur,res); // 不选 cur.push_back(a[k]); subsets(a,k+1,cur,res); // 选 cur.pop_back(); }
24 N 皇后(剪枝)
bool ok(int r,int c,std::vector<int>& col){ for(int i=0;i<r;i++) if(col[i]==c || abs(col[i]-c)==abs(i-r)) return false; return true; } void queen(int r,int n,std::vector<int>& col,int& cnt){ if(r==n){ cnt++; return; } for(int c=0;c<n;c++) if(ok(r,c,col)){ col[r]=c; queen(r+1,n,col,cnt); col[r]=-1; } }
25 DFS / BFS
// 邻接表 g；BFS 求无权最短路层数 void bfs(std::vector<std::vector<int>>& g,int s){ std::vector<int> dist(g.size(),-1); std::queue<int> q; q.push(s); dist[s]=0; while(!q.empty()){ int u=q.front(); q.pop(); for(int v:g[u]) if(dist[v]<0){ dist[v]=dist[u]+1; q.push(v); } } }
26 Dijkstra 最短路(非负权)
// 优先队列 O(E log V) using P=std::pair<int,int>; // {dist,node} std::vector<int> dijkstra(std::vector<std::vector<P>>& g,int s){ std::vector<int> d(g.size(),INT_MAX); d[s]=0; std::priority_queue<P,std::vector<P>,std::greater<P>> pq; pq.push({0,s}); while(!pq.empty()){ auto [du,u]=pq.top(); pq.pop(); if(du>d[u]) continue; for(auto [w,v]:g[u]) if(d[u]+w<d[v]){ d[v]=d[u]+w; pq.push({d[v],v}); } } return d; }

28 KMP 字符串匹配 O(n+m)
std::vector<int> build(std::string p){ // 前缀函数(next) std::vector<int> nx(p.size(),0); for(int i=1,j=0;i<(int)p.size();i++){ while(j&&p[i]!=p[j]) j=nx[j-1]; if(p[i]==p[j]) j++; nx[i]=j; } return nx; }
